## 1. 技术选型

### 1.1 前端

| 技术 | 版本 | 选型理由 |
|------|------|----------|
| **Next.js** | 15+ (App Router) | PWA 支持完善，SSR/SSG 可选，API Routes 可代理后端；React 生态最成熟 |
| **React** | 19 | Next.js 内置，并发特性提升交互流畅度 |
| **Tailwind CSS** | 4.x | 原子化 CSS，开发速度快，自带设计约束，移动端响应式友好 |
| **Dexie.js** | 4.x | IndexedDB 最成熟的封装库，Promise 风格 API，支持事务、索引、复合查询 |
| **Zustand** | 5.x | 比 Redux 轻一个数量级，API 极简，适合中等复杂度应用 |
| **Framer Motion** | 11.x | 闪卡翻转动画、页面过渡，轻量且声明式 |
| **React Hook Form** | 7.x | 表单验证（添加/编辑单词），性能好，依赖少 |
| **Zod** | 3.x | 运行时类型校验（导入数据验证、API 响应校验） |

#### 1.1.1 为什么不选其他方案

| 备选 | 淘汰原因 |
|------|----------|
| Vue + Nuxt | 生态插件不如 React 丰富（尤其 PWA / IndexedDB 工具链） |
| React Native | PWA 已经满足手机端需求，无需额外维护一套原生代码 |
| Redux / MobX | 状态复杂度不需要，Zustand 足够 |
| shadcn/ui | 可以考虑，但组件定制工作量大，不如 Tailwind 直接写 |

### 1.2 后端

| 技术 | 选型理由 |
|------|----------|
| **Node.js + Hono** | 轻量 Web 框架，边缘部署友好，TypeScript 原生支持，比 Express 快且更现代 |
| **PostgreSQL** | 成熟的关系型数据库，JSONB 支持灵活字段，同步冲突处理可靠 |
| **Prisma** | TypeScript 类型安全的 ORM，自动迁移，适合原型快速迭代 |
| **JWT (jose)** | 无状态认证，适合 PWA 离线场景 |
| **bcrypt** | 密码哈希，业界标准 |

#### 1.2.1 为什么不选其他方案

| 备选 | 淘汰原因 |
|------|----------|
| Express | 回调风格老旧，TypeScript 支持需额外配置 |
| MongoDB | 这个项目的数据模型是典型关系型（多对多关联），PostgreSQL 更适合 |
| SQLite (云端) | 单机存储，不适合多用户并发同步 |
| Supabase | 好选择，但自建更灵活且可控成本 |
| Next.js API Routes | 仅适合轻量 API，后续可能需要独立后端服务 |

### 1.3 开发工具

| 工具 | 用途 |
|------|------|
| TypeScript | 全栈类型安全 |
| ESLint + Prettier | 代码规范 |
| Vitest | 单元测试（快，与 Vite 生态一致） |
| Playwright | E2E 测试（PWA 离线场景） |
| GitHub Actions | CI/CD |

## 3. 数据库设计

### 3.1 云端 PostgreSQL (Prisma Schema)

```prisma
// ===== 用户 =====
model User {
  id            String    @id @default(uuid())
  email         String    @unique
  passwordHash  String
  streak        Int       @default(0)
  longestStreak Int       @default(0)
  lastCheckinAt DateTime?
  createdAt     DateTime  @default(now())
  updatedAt     DateTime  @updatedAt

  words         Word[]
  groups        Group[]
  tags          Tag[]
  studyRecords  StudyRecord[]
}

// ===== 单词 =====
model Word {
  id            String    @id @default(uuid())
  spanish       String
  chinese       String
  partOfSpeech  PartOfSpeech
  isStarred     Boolean   @default(false)
  audioUrl      String?
  groupId       String
  userId        String
  deletedAt     DateTime?        // 软删除
  createdAt     DateTime  @default(now())
  updatedAt     DateTime  @updatedAt

  group         Group     @relation(fields: [groupId], references: [id])
  user          User      @relation(fields: [userId], references: [id])
  examples      Example[]
  conjugation   Conjugation?
  srsState      SrsState?
  wordTags      WordTag[]
  studyRecords  StudyRecord[]
  relationsFrom WordRelation[]   @relation("RelationFrom")
  relationsTo   WordRelation[]   @relation("RelationTo")
}

// ===== 例句 =====
model Example {
  id           String       @id @default(uuid())
  spanish      String
  chinese      String
  source       ExampleSource
  sourceDetail String?
  wordId       String
  createdAt    DateTime     @default(now())

  word         Word         @relation(fields: [wordId], references: [id])
}

// ===== 动词变位 =====
model Conjugation {
  id          String  @id @default(uuid())
  wordId      String  @unique
  isRegular   Boolean @default(true)
  present     Json?   // {yo, tú, él, nosotros, vosotros, ellos}
  preterite   Json?
  imperfect   Json?
  future      Json?
  subjunctive Json?
  imperative  Json?
  conditional Json?

  word        Word    @relation(fields: [wordId], references: [id])
}

// ===== SRS 状态 =====
model SrsState {
  id             String   @id @default(uuid())
  wordId         String   @unique
  repetitions    Int      @default(0)
  easeFactor     Float    @default(2.5)
  interval       Float    @default(0)       // 天
  nextReviewAt   DateTime @default(now())
  lastReviewedAt DateTime?
  lastAnswer     Answer?

  word           Word     @relation(fields: [wordId], references: [id])
}

// ===== 分组 =====
model Group {
  id        String   @id @default(uuid())
  name      String
  source    GroupSource
  userId    String
  createdAt DateTime @default(now())

  user      User     @relation(fields: [userId], references: [id])
  words     Word[]
}

// ===== 标签 =====
model Tag {
  id     String @id @default(uuid())
  name   String
  color  String @default("#6366f1")
  userId String

  user   User   @relation(fields: [userId], references: [id])
  words  WordTag[]
}

// ===== 单词-标签关联 =====
model WordTag {
  wordId String
  tagId  String

  word   Word   @relation(fields: [wordId], references: [id], onDelete: Cascade)
  tag    Tag    @relation(fields: [tagId], references: [id], onDelete: Cascade)

  @@id([wordId, tagId])
}

// ===== 单词关联（同源/反义）=====
model WordRelation {
  wordId       String
  relatedWordId String
  relationType RelationType

  word         Word   @relation("RelationFrom", fields: [wordId], references: [id], onDelete: Cascade)
  relatedWord  Word   @relation("RelationTo", fields: [relatedWordId], references: [id], onDelete: Cascade)

  @@id([wordId, relatedWordId])
}

// ===== 学习记录 =====
model StudyRecord {
  id        String   @id @default(uuid())
  userId    String
  wordId    String
  mode      StudyMode
  direction Direction
  answer    Answer
  duration  Int?             // 秒
  createdAt DateTime @default(now())

  user      User     @relation(fields: [userId], references: [id])
  word      Word     @relation(fields: [wordId], references: [id])
}

// ===== 同步日志 =====
model SyncLog {
  id          String   @id @default(uuid())
  userId      String
  clientId    String           // 客户端标识
  lastSyncedAt DateTime
  changes     Json             // 本次同步的变更摘要
  createdAt   DateTime @default(now())
}

// ===== 枚举 =====
enum PartOfSpeech {
  NM   // 名词-阳性
  NF   // 名词-阴性
  NC   // 名词-阴阳同形
  V    // 动词
  ADJ  // 形容词
  ADV  // 副词
  PREP // 介词
  CONJ // 连词
  OTHER
}

enum ExampleSource {
  DICTIONARY
  CORPUS
  MANUAL
}

enum GroupSource {
  MANUAL
  FILE
  ANKI
}

enum RelationType {
  COGNATE
  ANTONYM
}

enum StudyMode {
  FLASHCARD
  SRS
}

enum Direction {
  ES_TO_ZH
  ZH_TO_ES
  LISTENING
}

enum Answer {
  REMEMBERED
  FORGOTTEN
}
```

### 3.2 本地 IndexedDB (Dexie.js Schema)

```typescript
// lib/db.ts

import Dexie, { type Table } from 'dexie';

export class SpanishVocabDB extends Dexie {
  words!: Table<Word, string>;
  examples!: Table<Example, string>;
  conjugations!: Table<Conjugation, string>;
  srsStates!: Table<SrsState, string>;
  groups!: Table<Group, string>;
  tags!: Table<Tag, string>;
  wordTags!: Table<WordTag, [string, string]>;
  wordRelations!: Table<WordRelation, [string, string]>;
  studyRecords!: Table<StudyRecord, string>;
  syncQueue!: Table<SyncQueueItem, string>;
  userMeta!: Table<UserMeta, string>;

  constructor() {
    super('SpanishVocabDB');
    this.version(1).stores({
      words: 'id, userId, groupId, partOfSpeech, isStarred, spanish, updatedAt',
      examples: 'id, wordId',
      conjugations: 'id, wordId',
      srsStates: 'id, wordId, nextReviewAt',
      groups: 'id, userId',
      tags: 'id, userId',
      wordTags: '[wordId+tagId], wordId, tagId',
      wordRelations: '[wordId+relatedWordId], wordId, relatedWordId',
      studyRecords: 'id, userId, wordId, createdAt',
      syncQueue: 'id, entityType, entityId, operation, createdAt',
      userMeta: 'key',
    });
  }
}

export const db = new SpanishVocabDB();
```

## 6. SRS 算法设计

### 6.1 SM-2 简化版算法

基于 Anki SM-2，但简化为两档（记得 / 不记得）：

```typescript
// lib/srs.ts

interface SrsState {
  repetitions: number;    // 连续「记得」次数
  easeFactor: number;     // 易度因子 (初始 2.5, 范围 1.3 - 2.5)
  interval: number;       // 当前间隔（天）
  nextReviewAt: Date;
  lastReviewedAt?: Date;
  lastAnswer?: 'remembered' | 'forgotten';
}

interface SrsResult {
  state: SrsState;
  isDue: boolean;         // 是否到期复习
}

export function calculateSrs(
  current: SrsState,
  answer: 'remembered' | 'forgotten'
): SrsState {
  const now = new Date();

  if (answer === 'forgotten') {
    // 忘记：重置进度
    return {
      repetitions: 0,
      easeFactor: Math.max(1.3, current.easeFactor - 0.2),
      interval: 0,                              // 当天内再出现
      nextReviewAt: addMinutes(now, 15),        // 15 分钟后重试
      lastReviewedAt: now,
      lastAnswer: 'forgotten',
    };
  }

  // 记得：计算新间隔
  let newInterval: number;
  let newRepetitions = current.repetitions + 1;

  switch (current.repetitions) {
    case 0:
      newInterval = 1 / 24;    // ≈ 1 小时（次日同时间）
      break;
    case 1:
      newInterval = 1;         // 1 天
      break;
    case 2:
      newInterval = 3;         // 3 天
      break;
    default:
      newInterval = Math.round(current.interval * current.easeFactor);
      // 上限 365 天
      newInterval = Math.min(newInterval, 365);
  }

  // 调整易度因子
  let newEase = current.easeFactor + 0.1;
  newEase = Math.min(newEase, 2.5);

  return {
    repetitions: newRepetitions,
    easeFactor: newEase,                        // 小幅提高
    interval: newInterval,
    nextReviewAt: addDays(now, newInterval),
    lastReviewedAt: now,
    lastAnswer: 'remembered',
  };
}

function addMinutes(date: Date, minutes: number): Date {
  return new Date(date.getTime() + minutes * 60000);
}

function addDays(date: Date, days: number): Date {
  const d = new Date(date);
  d.setDate(d.getDate() + Math.ceil(days));
  d.setHours(4, 0, 0, 0);      // 对齐到凌晨 4:00（避免深夜复习）
  return d;
}
```

### 6.2 复习队列生成

```typescript
export function getDueWords(
  allSrsStates: SrsState[],
  limit: number = 20
): SrsState[] {
  const now = new Date();

  return allSrsStates
    .filter(s => s.nextReviewAt <= now)  // 到期
    .sort((a, b) => {
      // 优先：逾期最久的 + 忘记过的 + 新词
      const aOverdue = now.getTime() - a.nextReviewAt.getTime();
      const bOverdue = now.getTime() - b.nextReviewAt.getTime();
      const aPriority = aOverdue + (a.lastAnswer === 'forgotten' ? 86400000 : 0);
      const bPriority = bOverdue + (b.lastAnswer === 'forgotten' ? 86400000 : 0);
      return bPriority - aPriority;
    })
    .slice(0, limit);
}
```

### 6.3 新词引入策略

- 用户首次学到某个单词（无 SrsState），标记为 **新词**
- 新词在「记得」1 次后进入 SRS 循环
- 每次复习会话可设置引入新词的上限（默认混合：80% 复习 + 20% 新词）

## 7. 同步引擎设计

### 7.1 同步策略

```
┌──────────────────────────────────────────────────┐
│                PUSH (本地 → 云端)                  │
│                                                    │
│  syncQueue 中有未推送记录                           │
│       │                                            │
│       ▼                                            │
│  POST /sync/push  { changes: [...] }               │
│       │                                            │
│       ├── 成功 → 清空已推送的 syncQueue 记录        │
│       └── 失败 → 增加 retryCount，等待下次重试      │
│                                                    │
│  触发时机：                                         │
│  - 本地有任何写操作后（debounce 2 秒）              │
│  - 应用从后台恢复时                                 │
│  - 用户手动点击「同步」                             │
└──────────────────────────────────────────────────┘

┌──────────────────────────────────────────────────┐
│                PULL (云端 → 本地)                  │
│                                                    │
│  获取上次同步时间戳 lastSyncedAt                     │
│       │                                            │
│       ▼                                            │
│  GET /sync/pull?since=lastSyncedAt                 │
│       │                                            │
│       ▼                                            │
│  对每条远程变更：                                   │
│  ┌─────────────────────────────────────────────┐  │
│  │ if 本地 updatedAt > 远程 updatedAt:          │  │
│  │     → 保留本地版本（本地更新）               │  │
│  │ if 本地 updatedAt < 远程 updatedAt:          │  │
│  │     → 用远程版本覆盖本地                     │  │
│  │ if 本地不存在该记录:                          │  │
│  │     → 写入新记录                             │  │
│  │ if 远程为 delete 且 本地 updatedAt < 远程:    │  │
│  │     → 删除本地记录                           │  │
│  └─────────────────────────────────────────────┘  │
│       │                                            │
│       ▼                                            │
│  更新 lastSyncedAt                                │
│                                                    │
│  触发时机：                                         │
│  - 应用启动时                                      │
│  - PUSH 成功后（双向同步）                          │
│  - 定时轮询（每 30 秒，仅当应用在前台）             │
└──────────────────────────────────────────────────┘
```

### 7.2 冲突处理策略

**简单策略：Last Write Wins (LWW)**

```
规则：以 updatedAt 时间戳为准，较新的覆盖较旧的

特殊处理：
- 同一实体在两台设备上都修改了 → 以 updatedAt 较新的为准
- 一台设备删除了某单词，另一台修改了它 → 删除优先（远程 delete + 本地 update → 执行 delete）
- 单词关联数据（examples, tags）视为单词的附属，随单词一起比较时间戳
```

这个策略对本应用足够——用户一般不会同时在两台设备上修改同一个单词。

### 7.3 离线行为

```typescript
// lib/sync.ts（伪代码）

class SyncEngine {
  private lastSyncedAt: Date | null;

  async push(): Promise<void> {
    if (!navigator.onLine) return;  // 离线跳过

    const queue = await db.syncQueue
      .orderBy('createdAt')
      .limit(100)
      .toArray();

    if (queue.length === 0) return;

    try {
      const res = await api.post('/sync/push', {
        changes: queue.map(item => ({
          entityType: item.entityType,
          entityId: item.entityId,
          operation: item.operation,
          payload: JSON.parse(item.payload),
          clientTimestamp: item.clientTimestamp,
        })),
      });

      // 删除已成功推送的记录
      await db.syncQueue.bulkDelete(queue.map(q => q.id));
    } catch (err) {
      console.error('Sync push failed, will retry', err);
    }
  }

  async pull(): Promise<void> {
    if (!navigator.onLine) return;

    try {
      const res = await api.get('/sync/pull', {
        since: this.lastSyncedAt?.toISOString(),
      });

      for (const change of res.changes) {
        await this.applyRemoteChange(change);
      }

      this.lastSyncedAt = new Date(res.serverTimestamp);
    } catch (err) {
      console.error('Sync pull failed', err);
    }
  }

  private async applyRemoteChange(change: RemoteChange): Promise<void> {
    const table = db.table(change.entityType + 's'); // 复数的表名映射
    const local = await table.get(change.entityId);

    if (change.operation === 'delete') {
      if (local && local.updatedAt < change.payload.updatedAt) {
        await table.delete(change.entityId);
      }
      return;
    }

    if (!local || local.updatedAt < change.payload.updatedAt) {
      await table.put(change.payload);
    }
    // else: 本地版本更新，保留本地
  }
}
```

## 8. 例句抓取引擎

### 8.1 多源抓取策略

```typescript
// server/src/services/sentenceService.ts

interface SentenceSource {
  name: string;
  type: 'DICTIONARY' | 'CORPUS';
  fetch(word: string): Promise<Example[]>;
}

class SentenceFetcher {
  private sources: SentenceSource[] = [
    new TatoebaSource(),         // 开源语料库，免费，REST API
    new OpenSubtitlesSource(),   // 电影字幕语料（备用）
    // 后续可扩展：
    // new SpanishDictSource(),   // 词典例句
    // new WordReferenceSource(), // 词典例句
  ];

  async fetchExamples(word: string): Promise<{
    examples: Example[];
    sources: string[];
  }> {
    const results: Example[] = [];
    const sources: string[] = [];

    for (const source of this.sources) {
      try {
        const examples = await source.fetch(word);
        if (examples.length > 0) {
          results.push(...examples.slice(0, 3)); // 每个源最多 3 条
          sources.push(source.name);
        }
      } catch (err) {
        console.warn(`Source ${source.name} failed for "${word}":`, err);
      }
    }

    return { examples: results.slice(0, 5), sources }; // 总计最多 5 条
  }
}
```
